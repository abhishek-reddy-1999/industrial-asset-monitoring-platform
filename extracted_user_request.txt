# PART 7 — CUSTOM JAVASCRIPT EDITOR

Display the available input variables, output variables and parameters.

Example:

function calculate(ctx, inputs) {
  const flow = inputs.flow;
  const power = inputs.power;
  const running = inputs.running;

  const valid =
    Number.isFinite(flow) &&
    flow > 0 &&
    Number.isFinite(power) &&
    power >= 0;

  return {
    specificEnergy: valid ? power / flow : null,
    isRunning: running === true || running === 1,
    validCalculation: valid
  };
}
